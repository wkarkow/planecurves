AutoLISP Code

(defun C:CURVENAME (/ basept xaxisangle cd sa sb sm om os divcount cosangle sinangle x0 y0 plineents segcount baseinfo paramstep
                     restartflag id iq dcl opt kw mode dpr dct dpt span eq_display xc yc skm a1 a2 x1 y1 x2 y2 fa fb fe)
  (setq id "Curve Name"
        iq "r = equation in display form"
        dcl nil
        opt nil
        kw 0
        mode 'polar          ; or 'param
        dpr "# of divisions around circle"
        dct 360
        dpt nil
        span 360
        kw (ca id iq dcl opt kw mode dpr dct dpt span)
        fa (get_real 2 "a (scale)" 1.0)
        fb (get_real 2 "b (description of effect)" 1.0)
        fe (get_keyword 0 "Real Imaginary" "Real or Imaginary values allowed?" "Real")
        eq_display (strcat "\na=" (rtos fa 2 4) ", b=" (rtos fb 2 4))
        ; define rval / xc / yc expressions here using fa, fb, fe ...
        skm '(T T T)
        a1 0
        a2 a1
        x1 (eval xc)
        y1 (eval yc)
        a2 '(+ a1 paramstep)
        x2 '(eval xc)
        y2 '(eval yc))
  (cc))
