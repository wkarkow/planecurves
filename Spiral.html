Spiral Curve

A single polar framework that recovers the Archimedean, Fermat, epispiral, parabolic, Galilean, lituus, hyperbolic, and logarithmic spirals by choice of nine parameters and also draws hybrid forms.

Spiral polar equation:
\[ r = a + b\theta^{c}\bigl(1 + d(\cos(f\theta))^{1/g}\bigr) + h\,e^{i\theta^{j}} \]

History and Applications

The spiral form1,2,3,4,5 has long held a powerful fascination across centuries of human thought, appearing as both a symbol of growth and an object of mathematical inquiry. From the steady unwinding of an Archimedean screw in ancient irrigation systems to the self-similar logarithmic spirals that govern the chambers of a nautilus shell and the arms of distant galaxies, spirals embody an intuitive link between angular motion and radial expansion. The present Spiral Curve is a generalization developed by the author beginning in 1988, with intermittent refinements over subsequent decades. Its single polar equation, \( r = a + b\theta^{c}\bigl(1 + d(\cos(f\theta))^{1/g}\bigr) + h\,e^{i\theta^{j}} \), was created specifically to encompass as many named classical spirals as possible within one accessible parametric framework. By suitable choice of the nine parameters \(a\) through \(j\) it reproduces the Archimedean, Fermat, epispiral, parabolic, Galilean, lituus, hyperbolic, and logarithmic spirals, and it readily generates hybrid forms for further exploration. Not every spiral fits this construction. The Atom spiral, Atzema spiral, Nielsen's spiral, and Poinsot's spiral, among others, require fundamentally different expressions. Certain visually spiral-like curves such as Fay's exponential butterfly also exhibit spiral characteristics in portions of their path but lie outside this family. Wikipedia’s List of spirals5 is a convenient comparative catalogue of named plane spirals, including members both inside and outside the present construction.

In engineering and the natural sciences the underlying ideas find direct application. Archimedean screws remain in use for pumping and conveying solids; logarithmic spirals appear in the design of certain antennas and waveguides; corrugated tubing and spiral springs rely on controlled radial growth for flexibility and energy storage. In nature, spiral growth governs phyllotaxis in plants, the coiling of mollusk shells, the structure of hurricanes, and the large-scale organization of galaxies.

(defun C:SPIRAL (/ basept xaxisangle cd sa sb sm om os divcount cosangle sinangle x0 y0 plineents segcount baseinfo paramstep divpt xspan restartflag
                   id iq dcl opt kw mode dpr dct dpt span eq_display xc yc skm a1 a2 x1 y1 x2 y2 fa fb fc fd fe ff fg fh fi fj fk fl fm fn p1)
  (setq id "Spiral Curve"
        iq ""
        dcl "SPIRAL"
        opt '("New Values" "Archimedes" "Epispiral" "Fermat" "Galilean" "Hyperbolic" "Lituus" "Logarithmic" "Parabolic")
        kw 0
        mode 'polar
        dpr "# of divisions around circle"
        dct 360
        dpt nil
        span 360
        kw (ca id iq dcl opt kw mode dpr dct dpt span)
        fk 1.0)
  (cond
   ((eq kw 0) ; New Values
    (princ "\nGeneral spiral: r=a+b·θ^c·(1+d·(cos(f·θ))^(1/g))+h·e^(i·θ^j)")
    (setq fa (get_real 0 "\na (constant radial offset)" 0.0)
          fb (get_real 0 "\nb (θ multiplier)" 0.5)
          fc (get_real 0 "\nc (θ exponent)" 1.0)
          fd (get_real 0 "\nd (cosine term multiplier)" 2.5)
          fe (get_real 0 "\nf (θ multiplier inside cosine)" 3.0)
          ff (get_real 0 "\ng (cosine exponent 1/g)" 1.5)
          fg (get_real 0 "\nh (exponential multiplier)" 0.15)
          fh (get_real 0 "\ni (θ multiplier inside exponential)" 0.08)
          fi (get_real 0 "\nj (θ exponent inside exponential)" 1.2)
          fj (get_keyword 0 "Real Imaginary" "Real or with Imaginary values allowed?" "Imaginary")))
   ((eq kw 1)
    (princ "\nArchimedean spiral: r=bθ")
    (setq p1 (get_boundpt "Point on curve" "1,0" (polar basept xaxisangle 1.0) basept nil nil nil nil nil nil)
          fl (+ (* (get_real 2 "# of extra revolutions to point" 0) (* 2.0 pi)) (- (angle basept p1) xaxisangle))
          fl (if (= fl 0) (+ fl (* 2.0 pi)) fl)
          fa 0
          fb (/ (distance basept p1) fl)
          fc 1 fd 0 fe 0 ff 1 fg 0 fh 1 fi 1 fj "Real"))
   ((eq kw 2)
    (princ "\nEpispiral: r = a·sec(nθ)")
    (setq p1 (get_boundpt "Point on curve" "1,0" (polar basept xaxisangle 1.0) basept nil nil nil nil nil nil)
          fe (get_real 2 "n (frequency multiplier inside cos)" 3.0)
          fl (+ (* (get_real 2 "# of extra revolutions to point" 0) (* 2.0 pi)) (- (angle basept p1) xaxisangle))
          fm (* (distance basept p1) (cos (* fe fl)))
          fa (- fm)
          fb fm
          fc 0 fd 1 ff -1 fg 0 fh 1 fi 1 fj "Real"))
   ((eq kw 3)
    (princ "\nFermat spiral: r=bθ^0.5")
    (setq p1 (get_boundpt "Point on curve" "1,0" (polar basept xaxisangle 1.0) basept nil nil nil nil nil nil)
          fl (+ (* (get_real 2 "# of extra revolutions to point" 0) (* 2.0 pi)) (- (angle basept p1) xaxisangle))
          fl (if (< (abs fl) 1e-8) (+ fl (* 2.0 pi)) fl)
          fa 0
          fb (/ (distance basept p1) (alt_expt fl 0.5 T))
          fc 0.5 fd 0 fe 0 ff 1 fg 0 fh 1 fi 1 fj "Real"))
   ((eq kw 4)
    (princ "\nGalilean spiral: r=bθ²")
    (setq p1 (get_boundpt "Point on curve" "1,0" (polar basept xaxisangle 1.0) basept nil nil nil nil nil nil)
          fl (+ (* (get_real 2 "# of extra revolutions to point" 0) (* 2.0 pi)) (- (angle basept p1) xaxisangle))
          fl (if (< (abs fl) 1e-8) (+ fl (* 2.0 pi)) fl)
          fa 0
          fb (/ (distance basept p1) (alt_expt fl 2.0 T))
          fc 2 fd 0 fe 0 ff 1 fg 0 fh 1 fi 1 fj "Real"))
   ((eq kw 5)
    (princ "\nHyperbolic spiral: r=b/θ")
    (setq p1 (get_boundpt "Point on curve" "1,0" (polar basept xaxisangle 1.0) basept nil nil nil nil nil nil)
          fl (+ (* (get_real 2 "# of extra revolutions to point" 0) (* 2.0 pi)) (- (angle basept p1) xaxisangle))
          fl (if (< (abs fl) 1e-8) (+ fl (* 2.0 pi)) fl)
          fa 0
          fb (* (distance basept p1) fl)
          fc -1 fd 0 fe 0 ff 1 fg 0 fh 1 fi 1 fj "Real"))
   ((eq kw 6)
    (princ "\nLituus spiral: r=bθ^(-0.5)")
    (setq p1 (get_boundpt "Point on curve" "1,0" (polar basept xaxisangle 1.0) basept nil nil nil nil nil nil)
          fl (+ (* (get_real 2 "# of extra revolutions to point" 0) (* 2.0 pi)) (- (angle basept p1) xaxisangle))
          fl (if (< (abs fl) 1e-8) (+ fl (* 2.0 pi)) fl)
          fa 0
          fb (/ (distance basept p1) (alt_expt fl -0.5 T))
          fc -0.5 fd 0 fe 0 ff 1 fg 0 fh 1 fi 1 fj "Real"))
   ((eq kw 7)
    (princ "\nLogarithmic spiral: r=g·e^(hθ)")
    (setq p1 (get_boundpt "Point on curve" "1,0" (polar basept xaxisangle 1.0) basept nil nil nil nil nil nil)
          fh (get_real 2 "h value for r = g·e^(hθ)" (- (log 2.0)))
          fa 0 fb 0 fc 1 fd 0 fe 0 ff 1
          fg (/ (distance basept p1) (exp (* fh (+ (* (get_real 2 "# of extra revolutions to point" 0) (* 2.0 pi)) (- (angle basept p1) xaxisangle)))))
          fi 1 fj "Real"))
   ((eq kw 8)
    (princ "\nParabolic spiral: r=b√θ")
    (setq p1 (get_boundpt "Point on curve" "1,0" (polar basept xaxisangle 1.0) basept nil nil nil nil nil nil)
          fl (+ (* (get_real 2 "# of extra revolutions to point" 0) (* 2.0 pi)) (- (angle basept p1) xaxisangle))
          fl (if (< (abs fl) 1e-8) (+ fl (* 2.0 pi)) fl)
          fa 0
          fb (/ (distance basept p1) (alt_expt fl 0.5 T))
          fc 0.5 fd 0 fe 0 ff 1 fg 0 fh 1 fi 1 fj "Real")))
  (setq eq_display (strcat "\na=" (rtos fa 2 4) " b=" (rtos fb 2 4) " c=" (rtos fc 2 4) " d=" (rtos fd 2 4) " f=" (rtos fe 2 4) " g=" (rtos ff 2 4)
                   " h=" (rtos fg 2 4) " i=" (rtos fh 2 4) " j=" (rtos fi 2 4))
        rval '(progn (setq fn (cos (* fe (eval a2)))) (if (eq fj "Real") (progn (setq restartflag (and fk (< (* fk fn) 0.0))) (setq fk fn) (if restartflag 1e14
              (+ fa (* fb (alt_expt (eval a2) fc T)) (+ 1.0 (* fd (alt_expt (cos (* fe (eval a2))) (/ 1.0 ff) T))) (* fg (exp (* fh (alt_expt (eval a2) fi T)))))))
              (+ fa (* fb (alt_expt (eval a2) fc nil)) (+ 1.0 (* fd (alt_expt (cos (* fe (eval a2))) (/ 1.0 ff) nil))) (* fg (exp (* fh (alt_expt (eval a2) fi nil)))))))
        xc '(* (eval rval) (cos (eval a2)))
        yc '(* (eval rval) (sin (eval a2)))
        skm '(T T T)
        a1 0
        a2 a1
        x1 (eval xc)
        y1 (eval yc)
        a2 '(+ a1 paramstep)
        x2 '(eval xc)
        y2 '(eval yc))
  (cc))
